--- --- {%- include jekyll/url.var -%} AWS S3 - Object Management

AWS Region

Parameters

##  Uploading an Object
aws s3 cp \
    --profile '' \
    '' 's3:///'

##  Listing Bucket Contents
# Top-level only.
aws s3 ls \
    --profile '' \
    's3:///'

# All objects recursively.
aws s3 ls \
    --profile '' \
    --recursive \
    's3:///'

##  Moving an Object to a Folder (Rename or Update Prefix)
aws s3 mv \
    --profile '' \
    's3:///' 's3:///'

##  Downloading an Object
# Using AWS CLI.
aws s3 cp \
    --profile '' \
    's3:///' ''

# Using `awscurl` (SigV4-signed HTTP Request).
### WARNING!!!
### For `legal90/awscurl`, do NOT use to download binary file. It will be corrupted!!!
### See: https://github.com/legal90/awscurl/issues/13
awscurl \
    --profile '' \
    --region '' \
    --service s3 \
    -H 'x-amz-content-sha256: e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855' \
    'https://s3..amazonaws.com//' \
    1> ''

##  Deleting an Object
aws s3 rm \
    --profile '' \
    's3:///'
