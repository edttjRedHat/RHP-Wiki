# AWS S3 - Simple Storage Service
## Bucket Management
<details><summary>Creating a Bucket</summary>

Bucket names are globally unique across all AWS accounts and regions. However,
a bucket is region-specific, hence create one per region to avoid cross-region
transfer costs.
```shell
__SHELL=0 \
    _AWS__S3__BUCKET_NAME__PFX='...bucketNamePfx...' \
    _AWS__S3__PUB_ACCESS=0 \
    _AWS__RESET_PROFILE=0 \
    _AWS__PROFILE=ocp \
    _AWS__ROLE_NAME_SFX=poweruser \
   x_AWS__SES_TO=3600 \
    _BW__NOTE_NAME='...bwNoteName...' \
    BW_SESSION="${BW_SESSION:+$([ -f "${BW_SESSION}" ] && cat "${BW_SESSION}" || echo "${BW_SESSION}")}" \
    BW_SESSION="$((bw status | grep -q '"status":"unlocked"') && echo "${BW_SESSION}" || bw unlock --raw || bw login --raw)" \
    AWS_REGION=us-east-1 \
    AWS_ACCOUNT_ID='...awsAccID...' \
    AWS_CONFIG_FILE="${HOME}/.aws/config" \
    AWS_SHARED_CREDENTIALS_FILE="${HOME}/.aws/credentials" \
    bash -o pipefail -O inherit_errexit -euc "$(cat - 0<<'cmdEOF'
        {
            ((_AWS__RESET_PROFILE)) && {
                # Clean up the AWS CLI profile.
                sed -i "/^\\[profile ${_AWS__PROFILE}\\]/,/^\\[/ {/^\\[profile ${_AWS__PROFILE}\\]/{d;b};/^\\[/"\!"d}" "${AWS_CONFIG_FILE}"
                sed -i "/^\\[${_AWS__PROFILE}\\]/,/^\\[/ {/^\\[${_AWS__PROFILE}\\]/{d;b};/^\\[/"\!"d}" "${AWS_SHARED_CREDENTIALS_FILE}"
            }
            klist -s || kinit
            aws-saml.py \
                --region "${AWS_REGION}" \
                --target-profile "${_AWS__PROFILE}" \
                --target-role "${AWS_ACCOUNT_ID}-${_AWS__ROLE_NAME_SFX}" \
                ${x_AWS__SES_TO:+--session-duration "${x_AWS__SES_TO}"}
            function aws () { command aws --profile "${_AWS__PROFILE}" "$@"; }
            export -f aws
            aws configure list --no-cli-pager
            aws sts get-caller-identity --no-cli-pager
        } 1>&2
        {
            typeset __shOpt="$(shopt -po xtrace)"; set +x
            [ -n "${BW_SESSION}" ] && bw sync || {
                echo 'You do NOT have an active and sync:ed BitWarden Session!!!' 1>&2
                exit 1
            }
            eval "${__shOpt}"; unset __shOpt
        }
        ((__SHELL)) && PROMPT_COMMAND='PS1="${PS1%\[DEBUG\] }[DEBUG] "' exec "${SHELL}" # Do NOT forget to exit this `DEBUG` session!!!

        typeset bwData= note=
        typeset s3BucketName="${_AWS__S3__BUCKET_NAME__PFX}--${AWS_REGION}"

        {
            typeset __shOpt="$(shopt -po xtrace)"; set +x
            bwData="$(bw get item "${_BW__NOTE_NAME}")" || {
                echo "You may NOT have access to BitWarden Note \`${_BW__NOTE_NAME}\`." 1>&2
                exit 1
            }
            { {
                bw encode 0<<<"${bwData}" |
                bw edit item "$(jq -cr '.id' 0<<<"${bwData}")" 1> /dev/null
            } || {
                echo "You do NOT have R/W access to BitWarden Note \`${_BW__NOTE_NAME}\`." 1>&2
                exit 1
            }; } && bw sync 1> /dev/null && bwData="$(bw get item "${_BW__NOTE_NAME}")"
            eval "${__shOpt}"; unset __shOpt
        }

        # Create bucket.
        aws s3api head-bucket \
            --region "${AWS_REGION}" \
            --bucket "${s3BucketName}" \
            1> /dev/null 2>&1 || {
            if [ "${AWS_REGION}" = "us-east-1" ]; then
                aws s3api create-bucket \
                    --region "${AWS_REGION}" \
                    --bucket "${s3BucketName}" \
                    --no-cli-pager
            else
                aws s3api create-bucket \
                    --region "${AWS_REGION}" \
                    --bucket "${s3BucketName}" \
                    --create-bucket-configuration "LocationConstraint=${AWS_REGION}" \
                    --no-cli-pager
            fi
        }

        # Block all public access as default policy.
        aws s3api put-public-access-block \
            --region "${AWS_REGION}" \
            --bucket "${s3BucketName}" \
            --public-access-block-configuration \
'BlockPublicAcls=true,'\
'IgnorePublicAcls=true,'\
"BlockPublicPolicy=$( ((_AWS__S3__PUB_ACCESS)) && echo false || echo true ),"\
"RestrictPublicBuckets=$( ((_AWS__S3__PUB_ACCESS)) && echo false || echo true )" \
            --no-cli-pager

        note="$(
            jq -cn \
                --arg awsRegion "${AWS_REGION}" \
                --arg s3URL "s3://${s3BucketName}/" \
                --arg httpPathURL  "https://s3.${AWS_REGION}.amazonaws.com/${s3BucketName}/" \
                --arg httpVhostURL "https://${s3BucketName}.s3.${AWS_REGION}.amazonaws.com/" \
                '{
                    awsRegions: {
                        $awsRegion: {
                            URLs: {
                                S3: $s3URL,
                                HTTP: {
                                    PathStyle: $httpPathURL,
                                    VirtualHostedStyle: $httpVhostURL
                                }
                            }
                        }
                    }
                }' |
            yq -p json -o yaml eval .
        )"

        # Update BitWarden Note.
        bwData="$(jq -r \
            --rawfile note <(
                yq eval-all '. as $item ireduce ({}; . * $item)' \
                    <(jq -r '.notes // ""' 0<<<"${bwData}") \
                    <(echo "${note}")
            ) \
            '.notes=$note' \
        0<<<"${bwData}")"
        bw encode 0<<<"${bwData}" | bw edit item "$(jq -cr '.id' 0<<<"${bwData}")" 1> /dev/null

        cat - 0<<txtEOF
--------------------------------------------------------------------------------
Bucket Created
--------------------------------------------------------------------------------
Bucket: ${s3BucketName}
Region: ${AWS_REGION}
URLs:
  S3:   s3://${s3BucketName}/
  HTTP:
    PathStyle:          https://s3.${AWS_REGION}.amazonaws.com/${s3BucketName}/
    VirtualHostedStyle: https://${s3BucketName}.s3.${AWS_REGION}.amazonaws.com/
--------------------------------------------------------------------------------
txtEOF

        true
cmdEOF
    )"; echo $?
```
</details>
<details><summary>Applying a Bucket Policy</summary>

Bucket policies grant access to IAM principals. For same-account access, either
a bucket policy or an IAM policy is sufficient (not both required). For
cross-account access, both sides must allow.

The suffix `/*` in the Resource ARN matches all objects at any depth including
those under folder prefixes (e.g., `win10/windows10.qcow2`).

The script below grants R/O access to one IAM user and R/W access to another.
```shell
__SHELL=0 \
    _AWS__S3__BUCKET_NAME__PFX='...bucketNamePfx...' \
    _AWS__S3__BUCKET_ACL__IAM_USR__PFX='...iamUsrPfx...' \
    _AWS__S3__PUB_ACCESS=0 \
    _AWS__IAM__ACC_KEY__NEW=0 \
    _AWS__RESET_PROFILE=0 \
    _AWS__PROFILE=ocp \
    _AWS__ROLE_NAME_SFX=poweruser \
   x_AWS__SES_TO=3600 \
    _BW__NOTE_NAME='...bwNoteName...' \
    _VAULT_MOUNT='...vaultMount...' \
    _VAULT_KEY_PATH='...vaultKeyPath...' \
    BW_SESSION="${BW_SESSION:+$([ -f "${BW_SESSION}" ] && cat "${BW_SESSION}" || echo "${BW_SESSION}")}" \
    BW_SESSION="$((bw status | grep -q '"status":"unlocked"') && echo "${BW_SESSION}" || bw unlock --raw || bw login --raw)" \
    AWS_REGION=us-east-1 \
    AWS_ACCOUNT_ID='...awsAccID...' \
    AWS_CONFIG_FILE="${HOME}/.aws/config" \
    AWS_SHARED_CREDENTIALS_FILE="${HOME}/.aws/credentials" \
    VAULT_ADDR='...vaultAddr...' \
    bash -o pipefail -O inherit_errexit -euc "$(cat - 0<<'cmdEOF'
        {
            ((_AWS__RESET_PROFILE)) && {
                # Clean up the AWS CLI profile.
                sed -i "/^\\[profile ${_AWS__PROFILE}\\]/,/^\\[/ {/^\\[profile ${_AWS__PROFILE}\\]/{d;b};/^\\[/"\!"d}" "${AWS_CONFIG_FILE}"
                sed -i "/^\\[${_AWS__PROFILE}\\]/,/^\\[/ {/^\\[${_AWS__PROFILE}\\]/{d;b};/^\\[/"\!"d}" "${AWS_SHARED_CREDENTIALS_FILE}"
            }
            klist -s || kinit
            aws-saml.py \
                --region "${AWS_REGION}" \
                --target-profile "${_AWS__PROFILE}" \
                --target-role "${AWS_ACCOUNT_ID}-${_AWS__ROLE_NAME_SFX}" \
                ${x_AWS__SES_TO:+--session-duration "${x_AWS__SES_TO}"}
            function aws () { command aws --profile "${_AWS__PROFILE}" "$@"; }
            export -f aws
            aws configure list --no-cli-pager
            aws sts get-caller-identity --no-cli-pager
        } 1>&2
        {
            typeset __shOpt="$(shopt -po xtrace)"; set +x
            [ -n "${BW_SESSION}" ] && bw sync || {
                echo 'You do NOT have an active and sync:ed BitWarden Session!!!' 1>&2
                exit 1
            }
            eval "${__shOpt}"; unset __shOpt
        }
        ((__SHELL)) && PROMPT_COMMAND='PS1="${PS1%\[DEBUG\] }[DEBUG] "' exec "${SHELL}" # Do NOT forget to exit this `DEBUG` session!!!

        typeset bwData= bwSubFld=
        typeset s3BucketName="${_AWS__S3__BUCKET_NAME__PFX}--${AWS_REGION}"
        typeset e1= e2=
        typeset -a iamUsrArr=()

        {
            typeset __shOpt="$(shopt -po xtrace)"; set +x
            bwData="$(bw get item "${_BW__NOTE_NAME}")" || {
                echo "You may NOT have access to BitWarden Note \`${_BW__NOTE_NAME}\`." 1>&2
                exit 1
            }
            { {
                bw encode 0<<<"${bwData}" |
                bw edit item "$(jq -cr '.id' 0<<<"${bwData}")" 1> /dev/null
            } || {
                echo "You do NOT have R/W access to BitWarden Note \`${_BW__NOTE_NAME}\`." 1>&2
                exit 1
            }; } && bw sync 1> /dev/null && bwData="$(bw get item "${_BW__NOTE_NAME}")"
            eval "${__shOpt}"; unset __shOpt
        }

        # IAM Users for S3 ACL.
        for e1 in "${_AWS__S3__BUCKET_ACL__IAM_USR__PFX}"--r{o,w}; do
            ((_AWS__S3__PUB_ACCESS)) && [[ "${e1}" == *--ro ]] && continue
            bwSubFld="cred.${e1}"
            iamUsrArr+=("${e1}")

            # Create IAM User.
            aws iam get-user \
                --user-name "${e1}" \
                --output text \
                --query 'User.Arn' \
                --no-cli-pager 2> /dev/null ||
            aws iam create-user \
                --user-name "${e1}" \
                --tags "$(
                    jq -cn \
                        --arg name "${e1}" \
                        '[{"Key": "Name", "Value": $name}]'
                )" \
                --no-cli-pager

            # Rotate IAM Access Key.
            ((_AWS__IAM__ACC_KEY__NEW)) && {
                typeset __shOpt="$(shopt -po xtrace)"; set +x
                # Delete stale Access Keys.
                for e2 in $(
                    aws iam list-access-keys --user-name "${e1}" |
                    jq -r \
                        --arg curKeyId "$(jq -r \
                            --arg fn__c "${bwSubFld}" \
                            '(.fields[]? | select(.name == $fn__c).value | fromjson.AccKeyId) // ""' \
                        0<<<"${bwData}")" \
                        '.AccessKeyMetadata[] | select(.AccessKeyId != $curKeyId) | .AccessKeyId'
                ); do aws iam delete-access-key --user-name "${e1}" --access-key-id "${e2}"; done
                # Generate new Access Key.
                bwData="$(jq -r \
                    --arg fn__c "${bwSubFld}" \
                    --rawfile fv__c <( set +x
                        aws iam create-access-key --user-name "${e1}" |
                        jq -cj '.AccessKey | {
                            AccKeyId: .AccessKeyId,
                            AccKeySecret: .SecretAccessKey
                        }'
                    true ) \
                    '.fields|=((. // []) | (
                        map(select(.name != $fn__c)) +
                        [{name: $fn__c, value: $fv__c, type: 1}]
                    ))' \
                0<<<"${bwData}")"
                eval "${__shOpt}"; unset __shOpt
            }

            # Configure AWS CLI profile.
            ( set +x
                typeset awsProfile="usr.${e1}"
                aws configure set \
                    aws_access_key_id \
                    "$(jq -r \
                        --arg fn__c "${bwSubFld}" \
                        '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeyId' \
                    0<<<"${bwData}")" \
                    --profile "${awsProfile}"
                aws configure set aws_secret_access_key \
                    "$(jq -r \
                        --arg fn__c "${bwSubFld}" \
                        '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeySecret' \
                    0<<<"${bwData}")" \
                    --profile "${awsProfile}"
            )
        done

        ((_AWS__IAM__ACC_KEY__NEW)) && {
            # Update BitWarden Note.
            {
                typeset __shOpt="$(shopt -po xtrace)"; set +x
                bwData="$(jq -r \
                    --rawfile fv__m <( set +x
                        printf '%s' "$(
                            typeset keyDate="$(date -u +"%Y-%m-%d %H:%M:%S %Z")"
                            typeset -p keyDate
                        )"
                    true ) '
                        (.fields[]? | select(.name == "metadata.keyDate")).value=$fv__m
                    ' \
                0<<<"${bwData}")"
                bw encode 0<<<"${bwData}" | bw edit item "$(jq -cr '.id' 0<<<"${bwData}")" 1> /dev/null
                eval "${__shOpt}"; unset __shOpt
            }

            # Update HashiCorp Vault.
            [ -z "${_VAULT_MOUNT}" ] || [ -z "${_VAULT_KEY_PATH}" ] || ( set +x
                typeset vaultData=

                vault token lookup &> /dev/null || {
                    echo 'Logging in to HashiCorp Vault...'
                    vault login 1> /dev/null
                } || {
                    echo "You may NOT have access to HashiCorp Vault at \`${VAULT_ADDR}\`." 1>&2
                    exit 1
                }

                for e1 in "${iamUsrArr[@]}"; do
                    vaultData="$(vault kv get -mount="${_VAULT_MOUNT}" -format=json "${_VAULT_KEY_PATH}")" || {
                        echo \
                            "You do NOT have access to HashiCorp Vault Secret"\
                            "\`${_VAULT_MOUNT}/${_VAULT_KEY_PATH}\` at \`${VAULT_ADDR}\`."\
                            1>&2
                        exit 1
                    }

                    [ "$(jq -cr \
                            --arg ki "AWS__S3__${e1}__AccKeyId" \
                            --arg ks "AWS__S3__${e1}__AccKeySecret" \
                            '
                                if . then {
                                    u: (.data.data[$ki] // ""),
                                    p: (.data.data[$ks] // "")
                                } else {u: "", p: ""} end
                            ' \
                    0<<<"${vaultData}")" = "$(
                        jq -cnr \
                            --arg ki "$(jq -r --arg fn__c "cred.${e1}" \
                                '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeyId' \
                            0<<<"${bwData}")" \
                            --arg ks "$(jq -r --arg fn__c "cred.${e1}" \
                                '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeySecret' \
                            0<<<"${bwData}")" \
                            '{u: $ki, p: $ks}'
                    )" ] || {
                        vault kv destroy -mount="${_VAULT_MOUNT}" -versions="$(
                            jq -r '.data.metadata.version' 0<<<"${vaultData}"
                        )" "${_VAULT_KEY_PATH}" &> /dev/null || true    # Ignore error (may not have delete permission).
                        {
                            jq -r \
                                --arg ki "AWS__S3__${e1}__AccKeyId" \
                                --arg vi "$(jq -r --arg fn__c "cred.${e1}" \
                                    '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeyId' \
                                0<<<"${bwData}")" \
                                --arg ks "AWS__S3__${e1}__AccKeySecret" \
                                --arg vs "$(jq -r --arg fn__c "cred.${e1}" \
                                    '.fields[]? | select(.name == $fn__c).value | fromjson.AccKeySecret' \
                                0<<<"${bwData}")" \
                                '.data.data | .[$ki]=$vi | .[$ks]=$vs' \
                            0<<<"${vaultData}" |
                            vault kv put -mount="${_VAULT_MOUNT}" "${_VAULT_KEY_PATH}" - 1> /dev/null
                        } || {
                            echo "You do NOT have R/W access to HashiCorp Vault secret \`${_VAULT_MOUNT}/${_VAULT_KEY_PATH}\`." 1>&2
                            exit 1
                        }
                    }
                done
            true )
        }

        aws s3api put-bucket-policy \
            --region "${AWS_REGION}" \
            --bucket "${s3BucketName}" \
            --policy "$(
                jq -cn \
                    --arg bucket "${s3BucketName}" \
                    --arg roUsr "arn:aws:iam::${AWS_ACCOUNT_ID}:user/${iamUsrArr[0]}" \
                    --arg rwUsr "arn:aws:iam::${AWS_ACCOUNT_ID}:user/${iamUsrArr[-1]}" \
                    --argjson pubAccess "$(
                        ((_AWS__S3__PUB_ACCESS)) && echo true || echo false
                    )" \
                    '{
                        Version: "2012-10-17",
                        Statement: [
                            {
                                Effect: "Allow",
                                Principal: {AWS: (if $pubAccess then
                                    [$rwUsr]
                                else
                                    [$roUsr, $rwUsr]
                                end)},
                                Action: "s3:ListBucket",
                                Resource: ("arn:aws:s3:::" + $bucket)
                            },
                            {
                                Effect: "Allow",
                                Principal: (if $pubAccess then
                                    "*"
                                else
                                    {AWS: $roUsr}
                                end),
                                Action: "s3:GetObject",
                                Resource: ("arn:aws:s3:::" + $bucket + "/*")
                            },
                            {
                                Effect: "Allow",
                                Principal: {AWS: $rwUsr},
                                Action: [
                                    "s3:GetObject", "s3:PutObject", "s3:DeleteObject",
                                    "s3:GetObjectTagging", "s3:PutObjectTagging"
                                ],
                                Resource: ("arn:aws:s3:::" + $bucket + "/*")
                            }
                        ]
                    }'
            )" \
            --no-cli-pager

        true
cmdEOF
    )"; echo $?
```
</details>


## Object Management
<details><summary>Uploading an Object</summary>

```shell
aws s3 cp \
    --profile '...awsProfile...' \
    '/path/to/file' 's3://bucket-name/optional/prefix/filename'
```
**Notes:**
  - For files over 5 GB, `aws s3 cp` automatically uses multi-part upload. The R/W IAM user needs `s3:ListBucket` on the bucket ARN (without `/*`) for
    multi-part uploads to succeed.
</details>
<details><summary>Listing Bucket Contents</summary>

```shell
# Top-level only.
aws s3 ls \
    --profile '...awsProfile...' \
    's3://bucket-name/'

# All objects recursively.
aws s3 ls \
    --profile '...awsProfile...' \
    --recursive \
    's3://bucket-name/'
```
</details>
<details><summary>Moving an Object to a Folder (Rename or Update Prefix)</summary>

```shell
aws s3 mv \
    --profile '...awsProfile...' \
    's3://bucket-name/object' 's3://bucket-name/optional/prefix/object'
```
**Notes:**
  - S3 has no real folders.
  - A "move" is a copy then delete.
  - For objects over 5 GB, `aws s3 mv` uses server-side `UploadPartCopy`. If
    the source and destination are in different regions and the machine running
    the CLI is behind a VPC S3 Gateway EndPoint, this will fail with
    `AccessDenied: VPC endpoints do not support cross-region requests`. As
    workaround, run the command from AWS CloudShell.
</details>
<details><summary>Downloading an Object</summary>

```shell
# Using AWS CLI.
aws s3 cp \
    --profile '...awsProfile...' \
    's3://bucket-name/optional/prefix/filename' '/path/to/file'

# Using `awscurl` (SigV4-signed HTTP Request).
### WARNING!!!
### For `legal90/awscurl`, do NOT use to download binary file. It will be corrupted!!!
### See: https://github.com/legal90/awscurl/issues/13
awscurl \
    --profile '...awsProfile...' \
    --region '<awsRegion>' \
    --service s3 \
    -H 'x-amz-content-sha256: e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855' \
    'https://s3.<awsRegion>.amazonaws.com/bucket-name/optional/prefix/filename' \
    1> '/path/to/file'
```
**Notes:**
  - The `legal90/awscurl` does not add `x-amz-content-sha256` automatically, but it is required by S3 private Bucket.
      - The SHA256 of empty body (GET) is always the same fixed value.
  - The `awscurl` buffers the full response in memory, hence it is not suitable for large files. Use `aws s3 cp`
    instead.
</details>
<details><summary>Deleting an Object</summary>

```shell
aws s3 rm \
    --profile '...awsProfile...' \
    's3://bucket-name/optional/prefix/filename'
```
</details>
