--- --- {%- include jekyll/url.var -%} HashiCorp Vault CLI Installation 
(
    osName="$(cat /etc/os-release | sed -nE '/^ID=/{s/^ID="([^"]+)"/\1/p;q}')"
    case ${osName} in
      (centos|rhel)
        dnf config-manager --add-repo 'https://rpm.releases.hashicorp.com/RHEL/hashicorp.repo'
        osMjrVer="$(awk -F'[=".]' '/^VERSION_ID=/{print $3}' /etc/os-release)"
        case ${osMjrVer} in
          (10)
            dnf config-manager --set-disabled hashicorp
            dnf --releasever=9 install -y --enablerepo=hashicorp vault
            ;;
          (*)
            dnf install -y vault
            ;;
        esac
        ;;
      (fedora)
        dnf config-manager --add-repo 'https://rpm.releases.hashicorp.com/fedora/hashicorp.repo'
        dnf install -y vault
        ;;
      (*)
        echo "Linux Distribution \`${osName}\` is not currently supported."
        ;;
    esac
)
